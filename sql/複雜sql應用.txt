 SELECT C.Name,AVG(C.CNT) as CNT FROM
  (SELECT B.Name, B.dt,COUNT(1) AS CNT FROM 
	  (SELECT A.Name as Name,A.dt,COUNT(A.UserId) AS Users FROM
		  ( SELECT T.TenancyName as Name, SUBSTRING(CONVERT(varchar(12),L.ExecutionTime,23),1,10) as dt,L.UserId
		  FROM [dbo].[AbpAuditLogs] L
		  INNER JOIN [dbo].[AbpTenants] T ON L.TenantId=T.Id
		  WHERE ExecutionTime>='2022-07-01' AND ExecutionTime<'2022-08-01') A
	  group by A.Name,A.dt,A.UserId)  B
  GROUP BY B.Name,B.dt) C
  GROUP BY C.Name
