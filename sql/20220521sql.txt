var qParas = (from p in _buManager.QueryUserBu.WhereIf(input.BuId!=null, x => x.BuId.Equals(input.BuId))
                      join b in _buManager.QueryBu  on p.BuId equals b.Id
                      join u in _buManager.QueryUsers.WhereIf(!string.IsNullOrEmpty(input.UserCode), x => x.UserName.Contains(input.UserCode))
                      .WhereIf(!string.IsNullOrEmpty(input.UserName), x => x.Name.Contains(input.UserName))
                      on p.UserId equals u.Id

                      select new UserBuOutput()
                      {
                          UserId = p.UserId,
                          BuId = p.BuId,
                          UserCode = u.UserName,
                          UserName = u.Name,
                          BuCode=b.Code,
                          BuName=b.Name

                      });
        var buList = qParas.ToList();
        var query = buList.GroupBy(t => new { t.UserId, t.UserCode, t.UserName })
       .Select(g => new UserBuOutput { UserId = g.Key.UserId, UserCode = g.Key.UserCode, UserName = g.Key.UserName, BuCode = string.Join(",", g.Select(s => s.BuCode).ToArray()), BuName = string.Join(",", g.Select(s => s.BuName).ToArray()) });

        var buQuery = query.AsQueryable();//主要用來爲PageBy服務的，如果不使用PageBy,可以簡化，直接toList()
        var buinfo = buQuery.OrderBy(input.Sorting).PageBy(input).ToList();

4.Enum 
public enum Roles
{
    Administrator = 1,
    Headquarters = 2,
    Branch = 3,
    Driver = 4,
    Client = 5
}
var roles = Enum.GetValues(typeof(Roles)).Cast<Roles>()
        .Select(r => new { Value = (int)r, Name = r.ToString() }).ToList();


you can then use this in your Linq query, for example:
var roles = Enum.GetValues(typeof(Roles)).Cast<Roles>()
        .Select(r => new { Value = (int)r, Name = r.ToString() }).ToList();

var users = from u in ApplicationUser
        join r in roles on u.Role equals r.Value
        select new {Name = u.Name, RoleId = u.Role, RoleDescription = r.Name} ;
